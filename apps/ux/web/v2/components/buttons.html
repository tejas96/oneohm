CSS Classes

Reference for all button classes.

/* Base */
.btn { ... }

/* Variants */
.btn-primary    /* Green background, white text */
.btn-secondary  /* Blue background, white text */
.btn-outline    /* Border only, transparent bg */
.btn-ghost      /* No border, transparent bg */
.btn-danger     /* Red background, white text */
.btn-success    /* Green background, white text */
.btn-warning    /* Amber background */
.btn-link       /* Text with underline */

/* Sizes */
.btn-sm  /* 32px height */
.btn-md  /* 40px height (default) */
.btn-lg  /* 48px height */
.btn-xl  /* 56px height */

/* Modifiers */
.btn-icon     /* Square, icon-only button */
.btn-pill     /* Fully rounded */
.btn-full     /* Full width */
.btn-loading  /* Shows spinner */

/* States */
:disabled     /* Opacity 0.5, cursor not-allowed */
:focus-visible/* Green outline ring */
